# NFSU2_PAD=hotplug: no controller at boot -> no game pad device; plugging
# one in is picked up by the game's own 32-frame rescan (0x5cfa10, enabled
# through [0x8364e8]); unplugging reports DIERR_INPUTLOST and the game
# releases the device; plugging in again re-creates it.
300   log --- boot without a controller ---
+0    peek 0x86ffb4                                  # slot 0: 0
+0    peek 0x870764                                  # input channels: 1 (keyboard)
+0    pad connect
+90   log --- plugged in ---
+0    peek 0x86ffb4                                  # slot 0: device
+0    peek 0x870764                                  # 2
+0    peek 0x8751e4                                  # channel 1 type 0x15 (gamepad)
+0    pad disconnect
+90   log --- unplugged ---
+0    peek 0x86ffb4                                  # 0 again
+0    peek 0x8751e0                                  # channel 1 slot: -1
+0    pad connect
+90   log --- plugged in again ---
+0    peek 0x86ffb4
+0    peek 0x8751e0
+0    peek 0x874fc8 2                                # default layout re-applied
+0    tappaduntil START MC_Background.fng 120 4000   # and the pad drives the menus
+200  tappad A
+0    waitfng GarageMain.fng 3000
+200  dump mainmenu
+30   quit
